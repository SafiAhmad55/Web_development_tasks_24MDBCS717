    <script>
        const form = document.getElementById("registrationForm");

        form.addEventListener("submit", function(event) {
            event.preventDefault();

            const name = document.getElementById("name").value.trim();
            const email = document.getElementById("mail").value.trim();
            const eventName = document.getElementById("event-selector").value;
            const detail = document.getElementById("detail");

            if (name === "" || email === "") {
                detail.value =
                    "Error: Please enter your student name and email.";
                detail.style.color = "red";
                return;
            }

            if (eventName === "") {
                detail.value = "Error: Please select an event.";
                detail.style.color = "red";
                return;
            }

            detail.value = "Congratulations, " + name +
                "! You have registered for " + eventName + ".";

            detail.style.color = "green";
        });
    </script>

    <script>
        $(document).ready(function() {

            $("#toggleinstruction").click(function() {
                $("#sh").toggle();
            });

            $("#toggleevent").click(function() {
                $("#eventDetails").slideToggle();
            });

            $("#addEvent").click(function() {
                const selectedEvent = $("#event-selector").val();

                if (selectedEvent === "") {
                    $("#jqueryMessage")
                        .text("Please select an event first.")
                        .fadeIn();
                    return;
                }

                $("#jqueryMessage").hide();

                const newItem = $("<li>")
                    .addClass("list-group-item")
                    .text(selectedEvent);

                $("#eventList").append(newItem);
            });

            $("#changeStyle").click(function() {
                $("#event_title").css({
                    "background-color": "black",
                    "color": "yellow"
                });
            });

            $("#event_title").mouseover(function() {
                $(this).css("color", "red");
            });

            $("#event_title").mouseout(function() {
                $(this).css("color", "blue");
            });

        });
    </script>
